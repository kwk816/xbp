10def ask_number(message, choices):
11    while True:
12        answer = input(message).strip()
13
14        if answer in choices:
15            return answer
16
17        print("その番号は選べないよ。もう一度入力してね。")

57def stage1():
58    print()
59    print("【STAGE 1：午前2時のコンビニ】")
60    print("バックヤードの扉を開ける暗証番号を探そう。")
61    line()
62
63    found_number = False
64
65    while not found_number:
66        print()
67        print("何を調べる？")
68        print("1：レジ")
69        print("2：おにぎりコーナー")
70        print("3：飲み物コーナー")
71        print("4：雑誌コーナー")
72        print("5：暗証番号を入力する")
73
74        choice = ask_number("番号を入力：",
75                            ["1", "2", "3", "4", "5"])
76
77        if choice == "1":
78            print()
79            print("レジには誰もいない。")
80            print("しかし、レシートが勝手に印刷された。")
81            print("レシートには「251」と書かれている。")
82            print("暗証番号のヒントを手に入れた！")
83            found_number = True
84
85        elif choice == "2":
86            print()
87            print("おにぎりがきれいに並んでいる。")
88            print("タクミ「帰れたら絶対ツナマヨ買う！」")
89
90        elif choice == "3":
91            print()
92            print("冷蔵庫の中は冷たい。")
93            print("ハル「この音、聞いたことがある気がする……。」")
94
95        elif choice == "4":
96            print()
97            print("雑誌の間にメモが挟まっていた。")
98            print("『答えはレジのレシートにある。』")
99            print("ミナト「レジを調べてみよう。」")
100
101        elif choice == "5":
102            if not found_number:
103                print("まだヒントが足りないようだ。")
104                continue
105
106            password = input("暗証番号：")
107
108            if password == "251":
109                print("ピッ！")
110                print("バックヤードの扉が開いた！")
111                break
112            else:
113                print("違うみたいだ……。")
114                continue
115
116        if found_number:
117            print()
118            print("暗証番号を入力できるようになった。")
119
120            choice2 = ask_number(
121                "扉の暗証番号を入力する？ 1：はい 2：まだ調べる ",
122                ["1", "2"]
123            )
124
125            if choice2 == "1":
126                while True:
127                    password = input("暗証番号：")
128
129                    if password == "251":
130                        print("ピッ！")
131                        print("バックヤードの扉が開いた！")
132                        return
133                    else:
134                        print("違うみたいだ。もう一度考えよう。")
